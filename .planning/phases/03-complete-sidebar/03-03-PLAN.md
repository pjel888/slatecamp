```typescript
const backdropTransition = prefersReducedMotion
  ? { duration: 0 }

  ? { duration: 0 }
  : { duration: 0.2, ease: 'easeOut' as const }
```

**Focus management (A11-01, A11-02):**
On expand (expanded becomes true), use useEffect to focus the first focusable element inside the panel:
```typescript
useEffect(() => {
  if (!expanded) return
  // Small delay to let AnimatePresence render the panel
  const timer = setTimeout(() => {
    const panel = document.querySelector('[role="dialog"][aria-label="Navigation sidebar"]')
    const firstFocusable = panel?.querySelector<HTMLElement>('a, button, [tabindex]:not([tabindex="-1"])')
    firstFocusable?.focus()
  }, 50)
  return () => clearTimeout(timer)
}, [expanded])
```

On collapse, focus returns to toggle button. This is already handled in SidebarProvider's Escape handler. For backdrop click, add focus return:
```typescript
const handleBackdropClick = () => {
  setExpanded(false)
  requestAnimationFrame(() => {
    toggleRef.current?.focus()
  })
}
```

**Render structure:**
```tsx
<AnimatePresence>
  {expanded && (
    <>
      {/* Backdrop -- SIBLING to panel, NOT parent (prevents click bubbling) */}
      <motion.div
        key="sidebar-backdrop"
        initial={{ opacity: 0 }}
        animate={{ opacity: 1 }}
        exit={{ opacity: 0 }}
        transition={backdropTransition}
        className="fixed inset-0 z-[39] bg-black/25"
        onMouseDown={handleBackdropClick}
        aria-hidden="true"
      />
      {/* Panel */}
      <motion.aside
        key="sidebar-panel"
        initial={prefersReducedMotion ? { opacity: 0 } : { x: -20, opacity: 0 }}
        animate={prefersReducedMotion ? { opacity: 1 } : { x: 0, opacity: 1 }}
        exit={prefersReducedMotion ? { opacity: 0 } : { x: -20, opacity: 0 }}
        transition={panelTransition}
        className="fixed inset-y-0 left-0 z-[40] w-[260px] bg-white border-r border-border shadow-md"
        role="dialog"
        aria-label="Navigation sidebar"
      >
        <SidebarPanel />
      </motion.aside>
    </>
  )}
</AnimatePresence>
```

CRITICAL: Backdrop and panel are SIBLINGS inside the fragment, not parent-child. Use `onMouseDown` on backdrop (not onClick) to avoid drag-release edge cases. Backdrop has `aria-hidden="true"`.

Full keyboard navigation (A11-05): Tab order within the panel is natural (links and buttons in DOM order). The inert attribute on main content (from SidebarProvider) prevents Tab from escaping to content behind the overlay.
